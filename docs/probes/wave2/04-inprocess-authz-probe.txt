# READ-ONLY: the script is piped on stdin (nothing written on the node); root only assembles a fresh ticket for root@pam with PVE's own assemble_ticket (no state written). 2026-10-15, stunt node anas-pve, PVE 9.2.21.
# 'nobody-probe@pve' does not exist: it stands in for a PVE user with no ACL entry (refused).

## Script
#!/usr/bin/perl
# READ-ONLY probe: runs PVE's own verifiers and ACL check in-process, as the
# calling uid (run as www-data to mirror a pveproxy worker). Writes nothing.
use strict; use warnings;
use Time::HiRes qw(time);
use PVE::RPCEnvironment;
use PVE::AccessControl;
my ($ticket) = @ARGV;
printf "euid=%d egid=%s\n", $>, $);
my $rpcenv = PVE::RPCEnvironment->init('pub');
my $t0 = time; $rpcenv->init_request(); my $t1 = time;
printf "init_request: %.2f ms\n", ($t1-$t0)*1000;
$t0 = time; $rpcenv->init_request(); $t1 = time;
printf "init_request (2nd, user.cfg version unchanged): %.2f ms\n", ($t1-$t0)*1000;
if ($ticket) {
  my $t = time;
  my ($u, $age) = eval { PVE::AccessControl::verify_ticket($ticket) };
  printf "verify_ticket: %s (%.2f ms)\n", ($@ ? "FAIL $@" : "user=$u age=$age"), (time-$t)*1000;
  eval { $rpcenv->check_user_enabled($u) }; print "check_user_enabled: ", ($@ ? "FAIL $@" : "ok\n");
}
for my $u ('root@pam', 'nobody-probe@pve') {
  for my $p ('Sys.Audit', 'Sys.Modify') {
    my $t = time;
    my $ok = $rpcenv->check($u, '/', [$p], 1);
    printf "check(%s, /, %s) = %s  (%.3f ms)\n", $u, $p, ($ok ? 'allowed' : 'refused'), (time-$t)*1000;
  }
}
my $csrf = PVE::AccessControl::assemble_csrf_prevention_token('root@pam');
my ($ts, $sig) = split /:/, $csrf, 2;
printf "csrf token shape: %s:<%d chars>\n", ($ts =~ /^[0-9A-F]{8}$/ ? '<8 hex ts>' : 'other'), length($sig);
my $t = time;
my $okc = PVE::AccessControl::verify_csrf_prevention_token('root@pam', $csrf, 1);
printf "verify_csrf(root\@pam, own token) = %s (%.3f ms)\n", ($okc ? 'ok' : 'FAIL'), (time-$t)*1000;
printf "verify_csrf(alice\@pve, root's token) = %s\n", (PVE::AccessControl::verify_csrf_prevention_token('alice@pve', $csrf, 1) ? 'ok' : 'refused');
printf "verify_csrf(root\@pam, garbage) = %s\n", (PVE::AccessControl::verify_csrf_prevention_token('root@pam', 'ZZZ', 1) ? 'ok' : 'refused');
